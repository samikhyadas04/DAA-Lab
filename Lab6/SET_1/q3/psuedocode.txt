
LCS(X, Y, m, n)
    let dp[0..m][0..n] be a new array

    for i = 0 to m
        for j = 0 to n
            if i == 0 or j == 0
                dp[i][j] = 0
            else if X[i-1] == Y[j-1]
                dp[i][j] = dp[i-1][j-1] + 1
            else
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])

    // Reconstruct the subsequence
    len = dp[m][n]
    i = m, j = n
    result = array of size len

    while i > 0 and j > 0
        if X[i-1] == Y[j-1]
            result[--len] = X[i-1]
            i--, j--
        else if dp[i-1][j] > dp[i][j-1]
            i--
        else
            j--

    return dp[m][n], result
